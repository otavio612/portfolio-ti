CREATE DATABASE academia; 
USE academia;

CREATE TABLE aluno (
	id INT AUTO_INCREMENT PRIMARY KEY, 
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(14) NOT NULL UNIQUE,
    idade INT NOT NULL CHECK (idade >= 14),
    peso DECIMAL (5,2),
    plano VARCHAR(50)
);
